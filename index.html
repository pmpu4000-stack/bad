function doOptions(e) { return ContentService.createTextOutput(""); } function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var params = JSON.parse(e.postData.contents); var action = params.action || "login"; // ----------------- 動作 1：登入驗證 + 自動讀取雲端紀錄 ----------------- if (action === "login") { var sheet = ss.getSheetByName("使用者清單"); var data = sheet.getDataRange().getValues(); var inputUser = String(params.username).trim(); var inputPass = String(params.password).trim(); var isAuthenticated = false; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === inputUser && String(data[i][1]).trim() === inputPass) { isAuthenticated = true; break; } } if (!isAuthenticated) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "帳號或密碼錯誤" })) .setMimeType(ContentService.MimeType.JSON); } // 驗證成功：讀取該使用者個人分頁的最新一筆紀錄 var userRecords = null; var userSheet = ss.getSheetByName(inputUser); if (userSheet) { var lastRow = userSheet.getLastRow(); if (lastRow > 1) { // 第 1 列是標題，第 2 列開始有歷史備份 var lastCellContent = userSheet.getRange(lastRow, 2).getValue(); // 第 2 欄為 JSON 資料 try { userRecords = JSON.parse(lastCellContent); } catch (err) { userRecords = null; } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "驗證成功", records: userRecords })).setMimeType(ContentService.MimeType.JSON); } // ----------------- 動作 2：儲存 LocalStorage 紀錄至個人分頁 ----------------- if (action === "saveRecord") { var username = String(params.username).trim(); var records = params.records; if (!username) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "未提供使用者帳號" })) .setMimeType(ContentService.MimeType.JSON); } var userSheet = ss.getSheetByName(username); if (!userSheet) { userSheet = ss.insertSheet(username); userSheet.appendRow(["備份時間", "LocalStorage 完整紀錄 (JSON)"]); } var timestamp = Utilities.formatDate(new Date(), "Asia/Taipei", "yyyy-MM-dd HH:mm:ss"); userSheet.appendRow([timestamp, JSON.stringify(records)]); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "紀錄已成功儲存至試算表！" })) .setMimeType(ContentService.MimeType.JSON); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
