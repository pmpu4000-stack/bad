function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const params = JSON.parse(e.postData.contents || "{}"); const action = params.action || "login"; // 登入驗證 if (action === "login") { const sheet = ss.getSheetByName("使用者清單"); if (!sheet) throw new Error("找不到【使用者清單】工作表"); const data = sheet.getDataRange().getValues(); const inputUser = String(params.username || "").trim(); const inputPass = String(params.password || "").trim(); let isAuthenticated = false; for (let i = 1; i < data.length; i++) { const username = String(data[i][0] || "").trim(); const password = String(data[i][1] || "").trim(); if (username === inputUser && password === inputPass) { isAuthenticated = true; break; } } if (!isAuthenticated) { return jsonResponse({ status: "error", message: "帳號或密碼錯誤" }); } let userRecords = []; const userSheet = ss.getSheetByName(inputUser); if (userSheet && userSheet.getLastRow() >= 2) { const jsonText = String(userSheet.getRange("B2").getValue() || "").trim(); if (jsonText) { try { userRecords = JSON.parse(jsonText); } catch (err) { userRecords = []; } } } return jsonResponse({ status: "success", message: "驗證成功", records: userRecords }); } // 儲存紀錄 if (action === "saveRecord") { const username = String(params.username || "").trim(); const records = params.records || {}; if (!username) { return jsonResponse({ status: "error", message: "未提供使用者帳號" }); } let userSheet = ss.getSheetByName(username); if (!userSheet) { userSheet = ss.insertSheet(username); userSheet.appendRow([ "備份時間", "完整JSON", "等級", "分數", "連勝", "最佳連勝", "作答次數", "答對次數", "正確率", "Level1正確率/精通", "Level2正確率/精通", "Level3正確率/精通", "Level4正確率/精通", "Level5正確率/精通", "Distinct答對字數", "學習中", "練習中", "精中" ]); } const now = new Date(); const todayStr = Utilities.formatDate(now, "Asia/Taipei", "yyyy-MM-dd"); const timestamp = Utilities.formatDate(now, "Asia/Taipei", "yyyy-MM-dd HH:mm:ss"); const jsonData = JSON.stringify(records); if (jsonData.length > 50000) { return jsonResponse({ status: "error", message: "資料量過大，無法儲存" }); } // 解析 spellAgent.v2 let agentData = {}; try { agentData = JSON.parse(records["spellAgent.v2"] || "{}"); } catch (err) { agentData = {}; } const levelInfo = agentData.level || {}; const lstat = agentData.lstat || {}; // 計算 distinct 答對字數 & 熟練度分布 let distinctCorrect = 0; let learningCount = 0; let practicingCount = 0; let masteredCount = 0; const stat = agentData.stat || {}; for (let wordId in stat) { const info = stat[wordId]; if (info.c > 0) distinctCorrect++; if (info.c === 1) learningCount++; if (info.c === 2) practicingCount++; if (info.c >= 3) masteredCount++; } const rowData = [ timestamp, jsonData, levelInfo.current || "", agentData.points || "", agentData.streak || "", agentData.best || "", agentData.attempts || "", agentData.correct || "", agentData.attempts ? (Math.round(agentData.correct / agentData.attempts * 100) + "%") : "", ((lstat["1"] && lstat["1"].c) || 0) + "/" + ((lstat["1"] && lstat["1"].a) || 0), ((lstat["2"] && lstat["2"].c) || 0) + "/" + ((lstat["2"] && lstat["2"].a) || 0), ((lstat["3"] && lstat["3"].c) || 0) + "/" + ((lstat["3"] && lstat["3"].a) || 0), ((lstat["4"] && lstat["4"].c) || 0) + "/" + ((lstat["4"] && lstat["4"].a) || 0), ((lstat["5"] && lstat["5"].c) || 0) + "/" + ((lstat["5"] && lstat["5"].a) || 0), distinctCorrect, learningCount, practicingCount, masteredCount ]; // 檢查第 2 列日期是否為今天 if (userSheet.getLastRow() >= 2) { const lastDate = userSheet.getRange(2, 1).getValue(); const lastDateStr = Utilities.formatDate(new Date(lastDate), "Asia/Taipei", "yyyy-MM-dd"); if (lastDateStr === todayStr) { // 覆寫今天的紀錄 userSheet.getRange(2, 1, 1, rowData.length).setValues([rowData]); } else { // 插入新紀錄 userSheet.insertRowBefore(2); userSheet.getRange(2, 1, 1, rowData.length).setValues([rowData]); } } else { // 沒有資料，直接新增 userSheet.insertRowBefore(2); userSheet.getRange(2, 1, 1, rowData.length).setValues([rowData]); } return jsonResponse({ status: "success", message: "紀錄已成功儲存（同一天覆寫最新紀錄）" }); } return jsonResponse({ status: "error", message: "未知的 action" }); } catch (err) { return jsonResponse({ status: "error", message: err.toString() }); } } // JSON 回應 function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
